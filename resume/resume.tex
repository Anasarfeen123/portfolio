% Each of the three heading macros below opens a minipage that stays open
% until \resumeItemListEnd closes it. That makes one full entry (heading +
% its bullet list) an atomic block for pagination purposes: LaTeX will never
% split a page between an entry's title and its own bullets — if the whole
% block doesn't fit in what's left of the page, it moves entirely to the
% next one instead of leaving an orphaned heading behind.
\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
  \begin{minipage}[t]{0.97\textwidth}
    \begin{tabular*}{\linewidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{minipage}[t]{0.97\textwidth}
    \begin{tabular*}{\linewidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{minipage}[t]{0.97\textwidth}
    \begin{tabular*}{\linewidth}{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-7pt}
}
